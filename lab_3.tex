% =========================================================
% 1. Objective
% =========================================================
\section{Objective}
To practice Linux file and directory management commands:
\begin{itemize}[noitemsep,topsep=1pt,leftmargin=1.2em]
  \item Create folders and nested paths (\cmd{mkdir}, \cmd{mkdir -p}).
  \item Create empty files (\cmd{touch}).
  \item Copy, move, and rename files (\cmd{cp}, \cmd{mv}).
  \item Delete files and directories (\cmd{rm}, \cmd{rm -r}, \cmd{rm -rf}).
\end{itemize}

% =========================================================
% 2. Theoretical Background
% =========================================================
\section{Theoretical Background}
Linux provides commands to create, duplicate, move, and delete files directly from the terminal without a GUI.

\subsection{Commands Practiced in This Lab}
\vspace{-2pt}
\begin{table}[h!]
\centering\footnotesize
\renewcommand{\arraystretch}{1.08}
\begin{tabularx}{\linewidth}{llp{5.5cm}X}
\toprule
\textbf{Command} & \textbf{Syntax} & \textbf{Simple Meaning / Purpose} & \textbf{Example} \\
\midrule
\cmd{mkdir} & \texttt{mkdir <name>} & Create new directory. & \texttt{mkdir docs} \\
\cmd{mkdir -p} & \texttt{mkdir -p <path>} & Create parent and child folders together. & \texttt{mkdir -p proj/src/utils} \\
\cmd{touch} & \texttt{touch <file>} & Create empty file or update timestamp. & \texttt{touch notes.txt} \\
\cmd{cp} & \texttt{cp <src> <dest>} & Copy file to new location/name. & \texttt{cp notes.txt backup.txt} \\
\cmd{mv} & \texttt{mv <src> <dest>} & Move or rename file/folder. & \texttt{mv old.txt new.txt} \\
\cmd{rm} & \texttt{rm <file>} & Delete file permanently. & \texttt{rm temp.log} \\
\cmd{rm -r} & \texttt{rm -r <folder>} & Delete folder and its contents recursively. & \texttt{rm -r old\_data} \\
\cmd{rm -rf} & \texttt{rm -rf <path>} & Force delete folder and files without prompt. & \texttt{rm -rf build} \\
\bottomrule
\end{tabularx}
\end{table}
\vspace{-6pt}

\subsection{Important Rules for File Operations}
\begin{itemize}[noitemsep,topsep=1pt,leftmargin=1.2em]
  \item \textbf{No Trash Bin:} \cmd{rm} deletes files immediately and permanently.
  \item \textbf{Parent Flag (\cmd{-p}):} Creates missing parent folders automatically.
  \item \textbf{Recursive Flag (\cmd{-r}):} Removes non-empty folders and their contents.
  \item \textbf{Force Flag (\cmd{-f}):} Ignores errors/prompts, deleting without warning.
\end{itemize}

% =========================================================
% 4. Procedure / Steps
% =========================================================
\section{Procedure / Steps}
Steps followed in this experiment:
\begin{enumerate}[noitemsep,topsep=1pt,leftmargin=1.4em]
  \item Create demo folder (\cmd{mkdir lab3\_demo}) and navigate inside (\cmd{cd}).
  \item Create folder \cmd{docs} (\cmd{mkdir}) and nested path \cmd{proj/src/utils} (\cmd{mkdir -p}).
  \item Create empty files \cmd{notes.txt} and \cmd{main.c} using \cmd{touch}.
  \item Copy \cmd{notes.txt} to \cmd{notes\_copy.txt} using \cmd{cp}.
  \item Move copy to \cmd{backup/} and rename to \cmd{notes\_v1.txt} using \cmd{mv}.
  \item Delete file with \cmd{rm}, and delete directories with \cmd{rm -r} and \cmd{rm -rf}.
\end{enumerate}

\vspace{2pt}\noindent\textbf{Explanation of Session 1 Results:}\par\vspace{1pt}
\begin{itemize}[noitemsep,topsep=1pt,leftmargin=1.2em]

\end{itemize}
