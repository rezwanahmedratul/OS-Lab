% =========================================================
% 1. Objective
% =========================================================
\section{Objective}
To practice file and text searching commands in Linux:
\begin{itemize}[noitemsep,topsep=1pt,leftmargin=1.2em]
  \item Find files by name and type using \cmd{find}.
  \item Search text inside files using \cmd{grep}.
  \item Use \cmd{grep} flags: case-insensitive (\cmd{-i}), line numbers (\cmd{-n}), invert (\cmd{-v}).
  \item Search folders recursively using \cmd{grep -r} and \cmd{grep -rn}.
\end{itemize}

% =========================================================
% 2. Theoretical Background
% =========================================================
\section{Theoretical Background}
Linux provides two main search tools: \cmd{find} locates files on disk, while \cmd{grep} searches text inside files.

\subsection{Commands Practiced in This Lab}
\vspace{-2pt}
\begin{table}[h!]
\centering\footnotesize
\renewcommand{\arraystretch}{1.06}
\begin{tabularx}{\linewidth}{llp{5.4cm}X}
\toprule
\textbf{Command} & \textbf{Syntax} & \textbf{Simple Meaning / Purpose} & \textbf{Example} \\
\midrule
\cmd{find} & \texttt{find <dir> -name "<pattern>"} & Find files matching name pattern. & \texttt{find . -name "*.log"} \\
\cmd{find -iname} & \texttt{find <dir> -iname "<pattern>"} & Case-insensitive name search. & \texttt{find . -iname "notes*"} \\
\cmd{find -type} & \texttt{find <dir> -type <f|d>} & Filter by file (\texttt{f}) or directory (\texttt{d}). & \texttt{find . -type d} \\
\cmd{grep} & \texttt{grep "<text>" <file>} & Search text inside file. & \texttt{grep "ERROR" server.log} \\
\cmd{grep -i} & \texttt{grep -i "<text>" <file>} & Case-insensitive text search. & \texttt{grep -i "error" server.log} \\
\cmd{grep -n} & \texttt{grep -n "<text>" <file>} & Show matching line numbers. & \texttt{grep -n "fail" server.log} \\
\cmd{grep -v} & \texttt{grep -v "<text>" <file>} & Invert search (lines without text). & \texttt{grep -v "DEBUG" app.log} \\
\cmd{grep -r} & \texttt{grep -r "<text>" <dir>} & Search recursively in all folders. & \texttt{grep -r "database" .} \\
\cmd{grep -rn} & \texttt{grep -rn "<text>" <dir>} & Recursive search with line numbers. & \texttt{grep -rn "TODO" .} \\
\bottomrule
\end{tabularx}
\end{table}
\vspace{-6pt}

% 4. Procedure / Steps

\section{Procedure / Steps}
\begin{enumerate}[noitemsep,topsep=1pt,leftmargin=1.4em]
  \item Navigate to practice directory \cmd{lab5\_search}.
  \item Find \texttt{.log} files using \cmd{find . -name "*.log"}.
  \item Find files case-insensitively using \cmd{find . -iname "readme*"}.
  \item Filter by type using \cmd{find . -type d} and \cmd{find . -type f}.
  \item Search for \texttt{"ERROR"} in log file using \cmd{grep}.
  \item Search case-insensitively (\cmd{grep -i}) and with line numbers (\cmd{grep -n}).
  \item Invert search using \cmd{grep -v} to exclude lines.
  \item Search recursively across directories using \cmd{grep -r} and \cmd{grep -rn}.
\end{enumerate}

% =========================================================
% 6. Conclusion
% =========================================================
\section{Conclusion}
Key takeaways:
\begin{enumerate}[noitemsep,topsep=1pt,leftmargin=1.4em]
  \item \cmd{find} locates files and directories by name, extension, and type.
  \item \cmd{grep} searches text inside files using flags like \cmd{-i}, \cmd{-n}, and \cmd{-v}.
  \item Recursive search (\cmd{grep -r}, \cmd{grep -rn}) quickly finds text across entire directory trees.
\end{enumerate}
