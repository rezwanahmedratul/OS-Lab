% =========================================================
% 1. Objective
% =========================================================
\section{Objective}
To practice terminal file viewers and text editors:
\begin{itemize}[noitemsep,topsep=1pt,leftmargin=1.2em]
  \item View file contents (\cmd{cat}, \cmd{head}, \cmd{tail}, \cmd{less}).
  \item Check and install editor packages (\cmd{which}, \cmd{sudo apt install}).
  \item Edit text with \cmd{nano} (save, exit, cut, paste, find, replace).
  \item Edit text with \cmd{vim} (modes, yank, delete, put, search, substitute).
\end{itemize}

% =========================================================
% 2. Theoretical Background
% =========================================================
\section{Theoretical Background}
Linux uses viewers (pagers) for reading files and command-line text editors for modifying files without a GUI.

\subsection{File Viewing Commands}
\vspace{-2pt}
\begin{table}[h!]
\centering\footnotesize
\renewcommand{\arraystretch}{1.06}
\begin{tabularx}{\linewidth}{llp{5.4cm}X}
\toprule
\textbf{Command} & \textbf{Syntax} & \textbf{Simple Meaning / Purpose} & \textbf{Example} \\
\midrule
\cmd{cat} & \texttt{cat <file>} & Print entire file content. & \texttt{cat config.txt} \\
\cmd{less} & \texttt{less <file>} & View file page by page (scrollable). & \texttt{less system.log} \\
\cmd{head} & \texttt{head -n <N> <file>} & Show first $N$ lines. & \texttt{head -n 5 sample.txt} \\
\cmd{tail} & \texttt{tail -n <N> <file>} & Show last $N$ lines (recent logs). & \texttt{tail -n 3 server.log} \\
\bottomrule
\end{tabularx}
\end{table}

% =========================================================
% 4. Procedure / Steps
% =========================================================
\section{Procedure / Steps}
Steps followed in this experiment:
\begin{enumerate}[noitemsep,topsep=1pt,leftmargin=1.4em]
  \item Display full log with \cmd{cat server.log}.
  \item View first 3 lines with \cmd{head -n 3} and last 3 lines with \cmd{tail -n 3}.
  \item View interactively with \cmd{less server.log} and exit with \key{q}.
  \item Verify and edit with nano: search, replace, cut, paste, save (\key{Ctrl+O}), and exit (\key{Ctrl+X}).
  \item Verify and edit with vim: switch modes (\key{i}, \key{Esc}), copy (\key{yy}), paste (\key{p}), cut (\key{dd}), substitute (\cmd{:\%s}), and save-quit (\cmd{:wq}).
\end{enumerate}
