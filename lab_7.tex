% =========================================================
% 1. Objective
% =========================================================
\section{Objective}
To practice everyday Linux terminal utilities:
\begin{itemize}[noitemsep,topsep=1pt,leftmargin=1.2em]
  \item Check user identity and system name (\cmd{whoami}, \cmd{who}, \cmd{hostname}).
  \item Display text, timestamps, and insert delays (\cmd{echo}, \cmd{date}, \cmd{sleep}).
  \item Clear terminal screen (\cmd{clear}) and read command manuals (\cmd{man}).
  \item View past commands (\cmd{history}).
  \item Create command shortcuts (\cmd{alias}) and view environment variables (\cmd{env}, \cmd{printenv}).
\end{itemize}

\begin{itemize}[noitemsep,topsep=1pt,leftmargin=1.2em]
  \item \textbf{Identity:} \cmd{whoami}, \cmd{who}, \cmd{hostname}, and \cmd{date} display user and system status.
  \item \textbf{Control \& Help:} \cmd{clear} clears screen, \cmd{echo} prints text, \cmd{sleep} pauses, \cmd{man} shows documentation.
  \item \textbf{Workflow:} \cmd{alias} creates shortcuts, \cmd{history} lists past commands, \cmd{env}/\cmd{printenv} check environment variables.
\end{itemize}

% 4. Procedure / Steps

\section{Procedure / Steps}
\begin{enumerate}[noitemsep,topsep=1pt,leftmargin=1.4em]
  \item Check user identity with \cmd{whoami} and system name with \cmd{hostname}.
  \item List logged-in users with \cmd{who} and display system time with \cmd{date}.
  \item Print message with \cmd{echo} and test delay with \cmd{sleep 2}.
  \item Clear terminal screen using \cmd{clear}.
  \item View manual page using \cmd{man echo}.
  \item Create and test command shortcut using \cmd{alias ll='ls -lh'}.
  \item Inspect environment variables using \cmd{printenv} and \cmd{env}.
  \item View command history using \cmd{history}.
\end{enumerate}

\section{Conclusion}
Key takeaways:
\begin{enumerate}[noitemsep,topsep=1pt,leftmargin=1.4em]
  \item \cmd{whoami}, \cmd{who}, \cmd{hostname}, and \cmd{date} inspect user and machine identity.
  \item \cmd{echo}, \cmd{sleep}, and \cmd{clear} control output and screen flow; \cmd{man} provides documentation.
  \item \cmd{alias} saves keystrokes with shortcuts; \cmd{history} recalls past commands.
  \item \cmd{env} and \cmd{printenv} inspect system environment variables.
\end{enumerate}
