\newcommand{\terminalwindow}[2]{%
\par\vspace{-3pt}%
{\centering
\begin{tikzpicture}
  % 1. Terminal text on main layer with JetBrains Mono font
  \node[
      text=termtext,
      font=\terminalfont\fontsize{7.2}{8.9}\selectfont,
      align=left,
      text width=\dimexpr\linewidth-0.85cm\relax,
      inner xsep=5pt,
      inner ysep=3.5pt,
      anchor=north west
  ] (body) at (0.26cm, -0.40cm) {#2};

  % Coordinates
  \coordinate (topleft) at (0, 0);
  \coordinate (bottomright) at ($(body.south -| \linewidth, 0) + (0, -3.5pt)$);
  \coordinate (titlecenter) at ($0.5*(topleft) + 0.5*(\linewidth, 0) + (0, -0.20cm)$);

  % 2. Background and Title bar on background layer
  \begin{scope}[on background layer]
    \fill[termbg, rounded corners=5pt] (topleft) rectangle (bottomright);
    \begin{scope}
      \clip[rounded corners=5pt] (topleft) rectangle (bottomright);
      \fill[termbar] (topleft) rectangle (\linewidth, -0.40cm);
      \draw[termborder, line width=0.5pt] (0, -0.40cm) -- (\linewidth, -0.40cm);
    \end{scope}
    \draw[termborder, line width=0.55pt, rounded corners=5pt] (topleft) rectangle (bottomright);
  \end{scope}

  % 3. Window control dots
  \fill[red!80!black] (0.30cm, -0.20cm) circle (0.055cm);
  \fill[yellow!80!black] (0.52cm, -0.20cm) circle (0.055cm);
  \fill[green!75!black] (0.74cm, -0.20cm) circle (0.055cm);

  % 4. Centered terminal title
  \node[
      text=gray!40!black,
      font=\sffamily\bfseries\fontsize{6.8}{7.2}\selectfont,
      anchor=center
  ] at (titlecenter) {#1};
\end{tikzpicture}\par}%
\vspace{-3pt}%
}
